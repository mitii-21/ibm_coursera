
## launches_by_site

## landing_rate_by_site
SELECT LaunchSite, COUNT(*) AS launches, AVG([Class]) AS landing_rate FROM launches GROUP BY LaunchSite ORDER BY landing_rate DESC
  LaunchSite  launches  landing_rate
  KSC LC 39A        22      0.772727
 VAFB SLC 4E        13      0.769231
CCAFS SLC 40        55      0.600000
